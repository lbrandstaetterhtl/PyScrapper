    Start["FilterWindow: User klickt Apply"]

4. AudioPlayer State Machine - Mit LibVLC & Playlist Management

Kompletter AudioPlayer Lifecycle mit allen States, Events und Übergängen. Zeigt LibVLC Initialisierung, Playlist-Loading, Track-Wechsel und Shuffle-Logic. 
flowchart TD
    Init["new AudioPlayer()"]
    Init --> VLCInit["_libVLC = new LibVLC Parameters:\n- '--verbose'\n- '--codec=avcodec'\n- '--network-caching=5000'"]
    VLCInit --> PlayerInit["_mediaPlayer = new MediaPlayer(_libVLC)"]
    PlayerInit --> EventReg["Register Events:\n- ESAdded\n- EndReached\n- MediaParsedChanged"]
    EventReg --> Standby(["STANDBY\n(Waiting for Command)"])
    
    Standby --> LoadCmd["LoadPlaylist(Playlist)"]
    LoadCmd --> StorePlaylist["_currentPlaylist = playlist\n_currentIndex = 0\n_playableItems = playlist.PlayableMediaIds"]
    StorePlaylist --> ShuffleInit{"IsShuffle?"}
    ShuffleInit -->|Ja| RandomizeList["Random.Shuffle(_playableItems)"]
    ShuffleInit -->|Nein| KeepOrder["Keep Original Order"]
    RandomizeList --> GotoPlay["PlayFile(0)"]
    KeepOrder --> GotoPlay
    
    GotoPlay --> FileLoad["_currentMediaId = _playableItems[0]"]
    FileLoad --> FindMedia["Find DownloadedMedia\nwhere Id == _currentMediaId"]
    FindMedia --> ValidPath["Check File.Exists(media.DownloadPath)"]
    ValidPath --> PathOK{"File Exists?"}
    PathOK -->|Nein| SkipTrack["AppLogger.LogNewMassage('File Missing')\nPlayNext()"]
    SkipTrack --> GotoPlay
    PathOK -->|Ja| CreateMedia["_currentMedia = new Media(filePath)"]
    CreateMedia --> SetPlayer["_mediaPlayer.Media = _currentMedia"]
    SetPlayer --> Play["_mediaPlayer.Play()"]
    Play --> Playing(["PLAYING\n(Media Running)"])
    
    Playing --> ESEvent["Event: ESAdded_Event\n(Elementary Stream)"]
    ESEvent --> CheckStream{"TrackType == Video?"}
    CheckStream -->|Ja| VideoFound["_hasVideo = true"]
    CheckStream -->|Nein| CheckAudio{"TrackType == Audio?"}
    CheckAudio -->|Ja| AudioFound["_hasAudio = true"]
    VideoFound --> NotifyVideo["Invoke VideoAvailableChanged(true)"]
    AudioFound --> Playing2["Continue Playing"]
    NotifyVideo --> Playing2["Continue Playing"]
    Playing2 --> Playing
    
    Playing --> PauseCmd["TogglePause()"]
    PauseCmd --> CheckPaused{_mediaPlayer.IsPaused?}
    CheckPaused -->|Nein| DoPause["_mediaPlayer.Pause()"]
    DoPause --> Paused(["PAUSED"])
    CheckPaused -->|Ja| Playing
    
    Paused --> ResumeCmd["TogglePause()"]
    ResumeCmd --> DoPlay["_mediaPlayer.Play()"]
    DoPlay --> Playing
    
    Playing --> StopCmd["Stop()"]
    StopCmd --> DoStop["_mediaPlayer.Stop()\n_currentIndex = -1\n_currentPlaylist = null"]
    DoStop --> Standby
    
    Playing --> NextTrack["Event: EndReached\n(Track Finished Naturally)"]
    NextTrack --> QueueWorker["System.Threading.ThreadPool.QueueUserWorkItem"]
    QueueWorker --> PlayNext["PlayNext()"]
    PlayNext --> IncrementIdx["_currentIndex += 1"]
    IncrementIdx --> CheckBoundary{"Index >= PlayableItems.Count?"}
    CheckBoundary -->|Ja| WrapAround["_currentIndex = 0 Loop"]
    CheckBoundary -->|Nein| Keep["Keep Index"]
    WrapAround --> FileLoad
    Keep --> FileLoad
    
    Playing --> ShuffleClick["ToggleShuffle()"]
    ShuffleClick --> ToggleFlag["_shuffle = !_shuffle"]
    ToggleFlag --> RemakeList["Update _playableItems Order\nif shuffle: Random.Shuffle\nelse: Original Order"]
    RemakeList --> ResetIdx["_currentIndex = 0"]
    ResetIdx --> Playing
    
    Paused --> Dispose["Dispose()"]
    Standby --> Dispose
    Dispose --> CleanVLC["_mediaPlayer?.Dispose()\n_libVLC?.Dispose()"]
    CleanVLC --> End(["END"])
